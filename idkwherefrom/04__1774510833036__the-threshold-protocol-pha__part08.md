            name: Identifier for the structure (e.g., "Electoral College")
            maintenance_cost: Cost in system-specific units
        """
        self.calcified_structures[name] = maintenance_cost

        """

        Returns:
            Total E in system-specific units
        """

    def calculate_optimal_resource_use(self) -> float:
        """
        Get optimal resource use for the system type.

        Returns:
            Optimal resource value in system-specific units
        """
        return self.unit_conversion[self.system_type]["optimal_resource"]
